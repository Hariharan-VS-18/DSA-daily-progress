int* productExceptSelf(int* nums, int numsSize, int* returnSize) {
    int *res= malloc(numsSize*sizeof(int));
    int *a=malloc(numsSize*sizeof(int));
    *returnSize =numsSize;
int pr=1,cr=1;
    for(int i=0;i<numsSize;i++){
        if(i==0){
            a[i]=pr;
            
        }
        else{
        pr*=nums[i-1];
        a[i]=pr;
        }
    }
    for(int i=numsSize-1;i>=0;i--){
        if(i==numsSize-1){
            res[i]=a[i]*cr;
        }
        else{
            cr*=nums[i+1];
            res[i]=a[i]*cr;
        }
    }
    return res;
}
