```
class MinStack(object):

    def __init__(self):
        self.stack=[]
        self.min_stack=[]

    def push(self, val):
        self.stack.append(val)
        if not self.min_stack or self.min_stack[-1]>=val:
            self.min_stack.append(val)
        

    def pop(self):
        if self.min_stack[-1]==self.stack[-1]:
            self.min_stack.pop()
        self.stack.pop()    
        

    def top(self):
        return self.stack[-1]
        

    def getMin(self):
        return self.min_stack[-1]

```
