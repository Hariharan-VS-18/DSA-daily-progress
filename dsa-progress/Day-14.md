## Solution

```
class Solution(object):
    def isValidSudoku(self, board):
        """
        :type board: List[List[str]]
        :rtype: bool
        """
        row=set()
        cl=set()
       
        for r in range(0,9,3):
            for cm in range(0,9,3):
                 box=set()
                 for i in range(3):
                    for j in range(3):
                        v=board[r+i][cm+j]
                        if v!='.':
                            if v in box:
                                return False
                                break
                            box.add(v)
        for i in range(len(board)):
            for j in range(len(board[i])):
                c=board[j][i]
                if c!='.':
                    if c in cl:
                        return False
                        break
                cl.add(c)        
            cl.clear()
        for i in range(len(board)):
            for j in range(len(board[i])):
                k=board[i][j]
                if k!='.':
                    if k in row:
                        return False
                        break
                row.add(k)        
            row.clear()        
        return True
```
