```
class Solution(object):
    def checkInclusion(self, s1, s2):
        """
        :type s1: str
        :type s2: str
        :rtype: bool
        """
        if len(s1)>len(s2):
            return False
        h={}
        m={}
        k=len(s1)
        n=len(s2)
        for i in s1:
            h[i]=h.get(i,0)+1
        for i in range(k):
            m[s2[i]]=m.get(s2[i],0)+1
        if m==h :
            return True
        else:
            for i in range(k,n):
                a=s2[i]
                b=s2[i-k]
                m[b]-=1
                if m[b]==0:
                    del m[b]
                m[a]=m.get(a,0)+1
                if h==m:
                    return True
                    break
        return False            
```
