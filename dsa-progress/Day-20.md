## Solution

```
int trap(int* height, int heightSize) {
    int l=0,r=heightSize-1,lmax=0,rmax=0,mizu=0;
    while(l<r){
        if(height[l]<height[r]){
            if(height[l]>=lmax){
                lmax=height[l];
            }else{
                mizu +=lmax-height[l];
            }
            l++;
        }
        else{
            if(rmax<=height[r]){
                rmax=height[r];
            }else{
                mizu +=rmax-height[r];
            }
            r--;
        }

    }
    return mizu;
}
```
