class Solution(object):
    def totalFruit(self, fruits):
        """
        :type fruits: List[int]
        :rtype: int
        """
        freq={}
        j=0
        m=0
        for i in range(len(fruits)):
            key=fruits[i]
            freq[key]=freq.get(key,0)+1
            while len(freq)>2:
                freq[fruits[j]]-=1
                if freq[fruits[j]]==0:
                    del freq[fruits[j]]
                j+=1
            m=max(m,i-j+1)        
        return m
