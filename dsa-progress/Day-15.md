class Solution(object):
    def longestConsecutive(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        if not nums:
            return 0            
        else:
            ct=set(nums)
            m=0
            for i in ct:
                if i-1 not in ct:
                    count=1
                    while i+1 in ct:
                        count+=1
                        i+=1
                    m=max(m,count)  
            return m
