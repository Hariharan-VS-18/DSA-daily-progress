## Solution

```
class Solution(object):
    def evalRPN(self, tokens):
        st=[]
        for i in tokens:
            if i in '+-*/':
                b=st.pop()
                a=st.pop()
                if i=='+':
                     st.append(a+b)
                elif i=='*':
                     st.append(a*b)
                elif i=='-':
                     st.append(a-b)
                else:
                     st.append(int(float(a)/b))           
            else:
                st.append(int(i))
        return st[-1]  
```
